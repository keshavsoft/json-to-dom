Quick Usage Example JavaScript Module 
import { buildSpecElement } from "../src/v2/build/buildSpecElement.js";

const spec = {
    tagName: "button",
    textContent: "Generate Invoice",
    attributes: { class: "px-4 py-2 bg-blue-600 text-white rounded shadow hover:bg-blue-700" },
    events: { click: () => alert("Invoice created!") }
};

// Generates native browser DOM element
const buttonNode = buildSpecElement({ inSpec: spec });
document.getElementById("app").appendChild(buttonNode);
