How It Works in Your App

dist/v7/min.js
// 1. Import the zero-dependency compiler import { buildSpecElement } from "https://keshavsoft.github.io/json-to-dom/dist/v7/min.js"; // 2. Declare your element specification const spec = { tagName: "button", textContent: "Save Voucher", attributes: { type: "button", class: "btn btn-primary px-4 py-2" }, events: { click: () => alert("Saved!") } }; // 3. Compile instantly into a native browser DOM node const domNode = buildSpecElement({ inSpec: spec }); document.getElementById("container").appendChild(domNode);
