How to Use Samples in Your Own Project

Include the standalone minified bundle in your HTML file via ES module or direct script tag, define your JSON specification, and call buildSpecElement: 

<div id="app"></div>

<!-- Option 1: Direct ES Module import -->
<script type="module">
    import { buildSpecElement } from "./dist/v3/min.js";

    const spec = {
        tagName: "div",
        attributes: { class: "p-4 bg-blue-50 text-blue-800 rounded-lg" },
        textContent: "Hello from declarative JSON!"
    };

    const node = buildSpecElement({ inSpec: spec });
    document.getElementById("app").appendChild(node);
</script>
