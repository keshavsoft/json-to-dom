How to Use Samples in Your Own Project

Include the standalone bundle with a single script tag, then call window.ks["json-to-dom"].buildSpecElement anywhere: 

<div id="app"></div>

<!-- Load standalone engine bundle -->
<script type="module" src="https://keshavsoft.github.io/json-to-dom/dist/v3/min.js"></script>

<!-- Run declarative rendering -->
<script type="module">
    const spec = {
        tagName: "div",
        attributes: { class: "p-4 bg-blue-50 text-blue-800 rounded-lg shadow-sm" },
        textContent: "Hello from declarative JSON!"
    };

    // Access engine via window.ks["json-to-dom"]
    const node = window.ks["json-to-dom"].buildSpecElement({ inSpec: spec });
    document.getElementById("app").appendChild(node);
</script>
